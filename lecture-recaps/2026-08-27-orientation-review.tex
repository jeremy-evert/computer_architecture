\begin{frame}{What counts as durable?}
\begin{itemize}\item Source file + clear name\item Repository history\item Reproducible command or workflow\item Evidence of what worked and what failed\item A short explanation another person can inspect\end{itemize}
\end{frame}
\begin{frame}{Failure is part of the evidence}
\begin{block}{When stuck}Expected $\rightarrow$ Actual $\rightarrow$ Tried\end{block}
Do not conceal an install/path/tool error. Record it, ask a bounded question, and preserve the useful output.
\end{frame}
\begin{frame}{Safety and judgment}
\begin{itemize}\item University workspaces are not public social media.\item Do not paste private data into an unapproved tool.\item Use the smallest adequate model.\item Ask a human to choose when there are divergent consequential paths.\end{itemize}
\end{frame}
